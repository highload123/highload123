
Для подключения к офисной сети из больщих и малых пердей, было решено организовать себе доступ, но так, что бы по возможности не заинтересовать интересующихся. 
Самым простым решением было использовать связку vpn-over-ssh. В качестве сервера использовалось FreeBSD 15.1-p2, как актуальная на тот момент.
Итак, на сервере:
1) OpenVPN
root@:/home/alm # cd /usr/ports/security/openvpn
root@:/usr/ports/security/openvpn # make install clean
2) Генерируем ключи сервера
root@:/usr/ports/security/openvpn # cd
root@:~ # easy-rsa init-pki
root@:~ # easy-rsa build-ca
root@:~ # easy-rsa gen-dh
root@:~ # easy-rsa gen-crl
и раскладываем ключи в /usr/locL/etc/openvpn/ в соответствии с конфигом
3) в /usr/locL/etc/openvpn/ кладём server.conf следующего содержания
root@:/usr/local/etc/openvpn # cat server.conf
local   127.0.0.1
port 1194
proto tcp
dev tun
ca /usr/local/etc/openvpn/pki/ca.crt
cert /usr/local/etc/openvpn/pki/issued/server.crt
dh /usr/local/etc/openvpn/pki/dh.pem
topology subnet
server 10.8.0.0 255.255.255.0
push "dhcp-option DNS 4.4.4.4"
push "dhcp-option DNS 1.1.1.1"
keepalive 10 120
persist-tun
status openvpn-status.log
verb 3
explicit-exit-notify 1
4) Генерируем ключи клиента
root@:~ # easyrsa build-client-full client nopass
и копируем их на клиента
5) В /etc/rc.conf.local добавляем
gateway_enable="YES"
openvpn_enable="YES"
openvpn_configfile="/usr/local/etc/openvpn/server.conf"
openvpn_dir="/usr/local/etc/openvpn"
6) Выполняем первое включение
root@:~ # service openvpn start
и, если это необходимо (ранее не включено)
root@:~ # sysctl -w net.inet.ip.forwarding=1
7) Создаём ключи клиента
root@:~ # cd /usr/local/etc/openvpn
root@:~ # setenv EASYRSA_TEMP_DIR "/var/tmp"
root@:~ # easy-rsa gen-req ogorod nopass
root@:~ # easyrsa gen-req ogorod
root@:~ # easyrsa sign-req client ogorod
где "ogorod" это хостнейм клиента
8) На стороне клиента используем следующий конфиг
client
dev tun
proto tcp
remote 127.0.0.1 1194
resolv-retry infinite
nobind
persist-key
persist-tun
remote-cert-tls server
cipher AES-256-GCM
auth SHA256
verb 3

<ca>
-----BEGIN CERTIFICATE-----
### ca.crt ###
-----END CERTIFICATE-----
</ca>

<cert>
-----BEGIN CERTIFICATE-----
### ogorod.crt ###
-----END CERTIFICATE-----
</cert>

<key>
-----BEGIN PRIVATE KEY-----
### ogorod.key ###
-----END PRIVATE KEY-----
</key>
9) Для подключения к установленному OpenVPN предварительно устанавливаем ssh-тунель:
C:\Users\user> ssh -L 1194:localhost:1194 ovpn
И подключаемся клиентом к серверу (по адресу 127.0.0.1:1194)
